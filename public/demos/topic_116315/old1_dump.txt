        function addToCart(dishId) {
            const existingItem = cart.find(item => item.dishId === dishId);
            if (existingItem) {
                existingItem.quantity++;
            } else {
                cart.push({ dishId, quantity: 1 });
            }
            renderDishes();
            const dish = dishes.find(d => d.id === dishId);
            showToast(`宸叉坊鍔?${dish.name}`);
        }

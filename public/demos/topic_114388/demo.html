🛤️ 完整使用流程 FLOW

从添加科目到生成计划,4 步搞定。

1

📚 添加科目 + 重点

手动输入 / 🤖 AI 推荐 / 📤 上传文档(PDF/Word/图片)自动提取。设置难度 + 模拟考分 + 标记弱项。

2

⏰ 设置可复习时段

7×24 热力图点选 / 多时段列表 / 演示数据一键填充。自动合并连续时段,统计每天总时长。

3

🚀 生成复习计划

scheduler 综合 4 维权重(难度/弱项/模拟考分/完成率),按艾宾浩斯曲线分配到时段。弱项优先。

4

📖 复习 + 反馈

标记完成 → 进度条 + 完成度同步; 标记 ❌不会 → 自动插入"今日补课"; 系统在 5 分钟前提醒。

🧮 核心算法 · 权重公式 CODE

每个重点的"复习优先级"由 4 维权重综合决定。

// scheduler.js · computePointWeight
function computePointWeight(subject, point) {
    let w = subject.difficulty || 2;        // 基础 = 难度 (1~3)

    // 1) 手动标记弱项
    if ((subject.weakPoints || []).includes(point)) w += 0.8;

    // 2) 模拟考分
    if (subject.simulatedScore != null) {
        if (subject.simulatedScore < 60)      w += 0.5;
        else if (subject.simulatedScore < 75) w += 0.2;
    }

    // 3) 完成率 < 30%
    const total  = (subject.points || []).length;
    const done   = (subject.completedPoints || []).length;
    if (total > 0 && done / total < 0.3)     w += 0.3;

    return w;
}

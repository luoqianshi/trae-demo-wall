SCARA机械臂结构

嵌套Group层级 -- createPlatformGripper()

ⓘ
SCARA机械臂通过 嵌套THREE.Group 实现关节层级结构。父级Group的变换会自动传递给所有子级，从而实现关节联动。每个关节通过修改 rotation 或 scale 属性控制运动。

🏗 Group嵌套层级树

platformGripperGroup // 位置: (3.5, 0.875, -0.8) 金属台表面 basePlate // 固定基座 0.3x0.08x0.3 bolt x4 // 底座螺栓 armSpindleGroup // [关节1] 主轴旋转组 -- rotation.y 控制转向 spindle // 主轴圆柱 高度=0.5 spindleFlange // 主轴法兰盘 gaugeBody // 气泵压力表 armHorizontalGroup // [关节2] 横向伸缩组 hBeam // 横向杆 -- scale.x 控制伸缩 (0.3~0.8) hMarker // 红色末端标记 armVerticalGroup // [关节3] 竖向伸缩组 -- position.x 跟随横向杆 vBeam // 竖向杆 -- scale.y 控制伸缩 (0.2~0.7) armGripperTip // [末端] 抓手组 -- position.y 跟随竖向杆 mountPlate // 安装板 gripperBody // 抓手主体 slider // 导轨滑块 platformFingerL // 左夹爪 -- position.x 控制开合 platformFingerR // 右夹爪 -- position.x 控制开合 tube x2 // 蓝色气管

自动转移状态机

updateAutoGripCycle() -- 14个状态

ⓘ
状态机由 animate() 每帧调用 updateAutoGripCycle() 驱动。每个状态通过 autoGripFrame 计数器控制过渡时机（通常60帧=1秒完成一个动作）。关节运动使用 平滑插值（diff * 0.05）实现缓动效果。

⏱ 平滑插值机制

✓
所有关节运动都使用 指数衰减插值：current += (target - current) * 0.05，这会产生自然的加速-减速效果（缓入缓出），使机械臂运动更接近真实物理行为。
平滑插值公式 算法 
// 每帧执行（约60fps）
currentValue += (targetValue - currentValue) * 0.05;

// 效果：
// 第1帧:  移动 5% 的剩余距离
// 第10帧: 移动 ~40% 的剩余距离
// 第30帧: 移动 ~78% 的剩余距离
// 第60帧: 移动 ~95% 的剩余距离（约1秒达到目标）

init() -- 场景初始化 JavaScript 
function init() {
    // 创建场景
    scene = new THREE.Scene();
    scene.background = new THREE.Color(0x1a2332);
    scene.fog = new THREE.Fog(0x1a2332, 15, 60);

    // 创建相机
    camera = new THREE.PerspectiveCamera(60, width / height, 0.1, 1000);
    camera.position.set(4, 4, 7);

    // 创建渲染器
    renderer = new THREE.WebGLRenderer({ antialias: true });
    renderer.shadowMap.enabled = true;
    renderer.toneMapping = THREE.ACESFilmicToneMapping;

    // 轨道控制器
    controls = new OrbitControls(camera, renderer.domElement);
    controls.enableDamping = true;

    // 按顺序创建场景元素
    setupLighting();
    createEnvironment();
    createConveyorSystem();      // 内部调用 createPlatformGripper()
    createMetalTable();
    createWorkpieces();
    createMaterialBox();
    connectWebSocket();

    animate();  // 启动动画循环
}
animate() -- 主动画循环 JavaScript 
function animate() {
    animationId = requestAnimationFrame(animate);
    const delta = 0.016;  // 固定时间步长 (~60fps)

    animateConveyor();           // 1. 传送带运动 + 工件移动
    updateGripperAnimation();     // 2. SCARA臂动画 + 状态机

    controls.update();               // 3. 轨道控制器阻尼更新
    renderer.render(scene, camera);   // 4. 渲染场景
}
updateGripperAnimation() -- SCARA臂动画更新 JavaScript 
function updateGripperAnimation() {
    // 1. 气泵压力动态变化
    if (vacuumPumpOn) {
        pumpPressure = Math.min(1.0, pumpPressure + 0.02);
    } else {
        pumpPressure = Math.max(0.0, pumpPressure - 0.03);
    }

    // 2. 抓手开合动画（指数插值）
    const target = gripperTargetOpen ? 1.0 : 0.0;
    gripperAnimProgress += (target - gripperAnimProgress) * 0.1;
    const offset = 0.008 + (0.03 - 0.008) * gripperAnimProgress;
    platformFingerL.position.x = -offset;
    platformFingerR.position.x = offset;

    // 3. 主轴旋转（平滑插值）
    const rotDiff = armTargetRotation - armSpindleGroup.rotation.y;
    armSpindleGroup.rotation.y += rotDiff * 0.05;

    // 4. 横向杆伸缩（平滑插值）
    const hScaleDiff = armTargetHorizontal - hBeam.scale.x;
    hBeam.scale.x += hScaleDiff * 0.05;
    armVerticalGroup.position.x = ARM_HORIZONTAL_MIN * hBeam.scale.x;

    // 5. 竖向杆伸缩（平滑插值）
    const vScaleDiff = armTargetVertical - vBeam.scale.y;
    vBeam.scale.y += vScaleDiff * 0.05;
    armGripperTip.position.y = -ARM_VERTICAL_MIN * vBeam.scale.y;

    // 6. 工件跟随抓手末端
    if (grippedWorkpiece) {
        armGripperTip.getWorldPosition(tipPos);
        grippedWorkpiece.mesh.position.set(tipPos.x, tipPos.y - 0.08, tipPos.z);
    }

    // 7. 驱动自动转移状态机
    updateAutoGripCycle();
    updateDeviceStatus();
}
animateConveyor() -- 传送带动画 JavaScript 
function animateConveyor() {
    if (!conveyorRunning) return;

    // 滚筒旋转动画
    conveyorRollers.forEach(roller => {
        roller.rotation.z += 0.05;
    });

    // 工件沿传送带移动
    workpieces.forEach(wp => {
        if (!wp.gripped && !wp.onMetalTable) {
            wp.mesh.position.x += 0.008;  // 每帧移动0.008单位

            // 到达传送带末端 -> 滑落到金属台
            if (wp.mesh.position.x > 3.5) {
                wp.onMetalTable = true;
                conveyor.remove(wp.mesh);
                scene.add(wp.mesh);
                wp.mesh.position.set(3.8 + random, 0.9, -0.8);
            }
        }
    });
}
doReleaseWorkpiece() -- 释放工件到物料箱 JavaScript 
function doReleaseWorkpiece() {
    if (!grippedWorkpiece) return;
    const wp = grippedWorkpiece;

    // 获取抓手末端和物料箱的世界坐标
    armGripperTip.getWorldPosition(tipPos);
    materialBox.getWorldPosition(boxPos);

    // 判断是否在物料箱范围内
    if (dx < 0.6 && dz < 0.6) {
        // 放入物料箱：从conveyor移除，添加到scene
        conveyor.remove(wp.mesh);
        scene.add(wp.mesh);
        wp.mesh.position.set(
            boxPos.x + random() * 0.3,
            boxPos.y + 0.02 + boxContents.length * 0.06,
            boxPos.z + random() * 0.2
        );
        boxContents.push(wp);
    }

    grippedWorkpiece = null;
}
findWorkpieceAtGripper() -- 检测抓手附近工件 JavaScript 
function findWorkpieceAtGripper() {
    platformGripperGroup.getWorldPosition(basePos);

    let closest = null;
    let closestDist = Infinity;

    workpieces.forEach(wp => {
        if (wp.gripped) return;
        wp.mesh.getWorldPosition(wpPos);

        const dist = Math.sqrt(dx*dx + dz*dz);

        // 检测条件：在抓手可达范围内 + 在金属台上
        if (dist >= ARM_HORIZONTAL_MIN - 0.1
            && dist <= ARM_HORIZONTAL_MAX + 0.1
            && dy < 0.5
            && Math.abs(wpPos.z - (-0.8)) < 0.5) {
            if (dist < closestDist) {
                closestDist = dist;
                closest = wp;
            }
        }
    });

    return closest;
}

# 知识库重构开发计划

> 创建时间：2026-07-06
> 目标：基于 Neo4j + 工作流系统，重构知识入库模块

---

## 一、开发目标

### 1.1 核心目标

| 目标 | 说明 |
|------|------|
| **信息类型分离** | 事件、观点、案例、思维模型、描述性知识分开存储 |
| **因果链支持** | 通过 Neo4j 实现事件因果链查询 |
| **人物画像** | 实体-事件关系支持人物完整画像 |
| **可视化入库** | 通过工作流系统配置入库流程 |
| **混合检索** | 图遍历 + 向量召回结合 |

### 1.2 不改动的部分

| 模块 | 说明 |
|------|------|
| 思维模型入库 | 保持现有设计（thinking_model 表） |
| 工作流引擎 | 保持现有设计（FlowGraphExecutor） |
| 向量存储 | 保持 pgvector 扩展 |

## 二、数据表修正

### 2.1 新增表

#### events 表（事件主表）

```sql
CREATE TABLE events (
    id VARCHAR(36) PRIMARY KEY,
    kb_id VARCHAR(36) NOT NULL,
    doc_id VARCHAR(36),
    
    -- 事件核心
    action VARCHAR(100) NOT NULL,       -- 动作类型（收购/发布/裁员...）
    result TEXT,                         -- 结果/影响
    event_time DATE,                     -- 事件发生时间
    assertion_time DATE,                 -- 信息发布时间
    event_type VARCHAR(50),              -- 事件分类
    
    -- 附属物（用户强调）
    artifacts JSONB,                     -- 产出物、数据、媒体
    
    -- 证据
    evidence JSONB,                      -- 支撑证据
    
    -- Neo4j 节点 ID（用于关联）
    neo4j_id VARCHAR(36),                -- 对应 Neo4j Event 节点 ID
    
    -- 元数据
    source_text TEXT,                    -- 原文片段
    confidence DECIMAL(3,2),             -- 置信度
    create_time TIMESTAMP,
    is_delete INT DEFAULT 0
);

-- 索引
CREATE INDEX idx_events_kb ON events(kb_id);
CREATE INDEX idx_events_doc ON events(doc_id);
CREATE INDEX idx_events_action ON events(action);
CREATE INDEX idx_events_time ON events(event_time);
CREATE INDEX idx_events_neo4j ON events(neo4j_id);
```

#### entities 表（实体主表）

```sql
CREATE TABLE entities (
    id VARCHAR(36) PRIMARY KEY,
    kb_id VARCHAR(36) NOT NULL,
    
    -- 实体核心
    name VARCHAR(200) NOT NULL,
    type VARCHAR(50) NOT NULL,          -- 人物/组织/产品/地点/概念
    aliases JSONB,                       -- 别名列表
    description TEXT,                    -- 描述
    
    -- 事实（JSON数组）
    facts JSONB,                         -- 关于该实体的客观事实
    
    -- Neo4j 节点 ID
    neo4j_id VARCHAR(36),
    
    -- 统计字段
    event_count INT DEFAULT 0,           -- 参与事件数
    first_seen DATE,
    last_seen DATE,
    
    create_time TIMESTAMP,
    is_delete INT DEFAULT 0
);

-- 索引
CREATE INDEX idx_entities_kb ON entities(kb_id);
CREATE INDEX idx_entities_name ON entities(name);
CREATE INDEX idx_entities_type ON entities(type);
CREATE INDEX idx_entities_neo4j ON entities(neo4j_id);

-- 全文搜索索引（用于实体模糊匹配）
CREATE INDEX idx_entities_name_fts ON entities USING gin(to_tsvector('simple', name));
```

#### event_relations 表（事件关系表）

```sql
CREATE TABLE event_relations (
    id VARCHAR(36) PRIMARY KEY,
    kb_id VARCHAR(36) NOT NULL,
    
    -- 关系核心
    source_event_id VARCHAR(36) NOT NULL,
    target_event_id VARCHAR(36) NOT NULL,
    relation_type VARCHAR(50) NOT NULL,  -- CAUSED/TRIGGERED/FOLLOWED/CONCURRENT...
    
    -- 关系属性
    mechanism TEXT,                      -- 因果机制
    confidence DECIMAL(3,2),
    
    -- Neo4j 关系 ID
    neo4j_rel_id VARCHAR(36),
    
    create_time TIMESTAMP,
    
    FOREIGN KEY (source_event_id) REFERENCES events(id),
    FOREIGN KEY (target_event_id) REFERENCES events(id)
);

-- 索引
CREATE INDEX idx_rel_source ON event_relations(source_event_id);
CREATE INDEX idx_rel_target ON event_relations(target_event_id);
CREATE INDEX idx_rel_type ON event_relations(relation_type);
```

#### entity_event_roles 表（实体-事件角色表）

```sql
CREATE TABLE entity_event_roles (
    id VARCHAR(36) PRIMARY KEY,
    kb_id VARCHAR(36) NOT NULL,
    
    -- 关联
    entity_id VARCHAR(36) NOT NULL,
    event_id VARCHAR(36) NOT NULL,
    
    -- 角色
    role VARCHAR(50) NOT NULL,           -- INITIATED/TARGETED/PARTICIPATED/AFFECTED_BY...
    context TEXT,                        -- 角色上下文
    impact_type VARCHAR(50),             -- 影响类型
    impact_degree VARCHAR(20),           -- 影响程度（high/medium/low）
    
    -- Neo4j 关系 ID
    neo4j_rel_id VARCHAR(36),
    
    confidence DECIMAL(3,2),
    create_time TIMESTAMP,
    
    FOREIGN KEY (entity_id) REFERENCES entities(id),
    FOREIGN KEY (event_id) REFERENCES events(id)
);

-- 索引
CREATE INDEX idx_eer_entity ON entity_event_roles(entity_id);
CREATE INDEX idx_eer_event ON entity_event_roles(event_id);
CREATE INDEX idx_eer_role ON entity_event_roles(role);
```

#### viewpoints 表（观点表）

```sql
CREATE TABLE viewpoints (
    id VARCHAR(36) PRIMARY KEY,
    kb_id VARCHAR(36) NOT NULL,
    doc_id VARCHAR(36),
    
    -- 观点核心
    entity_id VARCHAR(36),               -- 观点持有者
    viewpoint_content TEXT NOT NULL,     -- 观点内容
    topic VARCHAR(200),                  -- 观点主题
    
    -- 认识论标签
    claim_type VARCHAR(50),              -- fact/opinion/prediction/marketing
    confidence DECIMAL(3,2),
    
    -- 时间
    assertion_time DATE,
    
    -- 关联事件
    related_event_ids JSONB,
    
    -- 关系类型（对事件的态度）
    relation_to_events JSONB,            -- [{event_id, relation: SUPPORTS/OPPOSES}]
    
    -- 来源
    source_text TEXT,
    
    -- 向量（pgvector）
    content_embedding vector(1024),
    
    create_time TIMESTAMP,
    is_delete INT DEFAULT 0,
    
    FOREIGN KEY (entity_id) REFERENCES entities(id)
);

-- 索引
CREATE INDEX idx_viewpoints_kb ON viewpoints(kb_id);
CREATE INDEX idx_viewpoints_entity ON viewpoints(entity_id);
CREATE INDEX idx_viewpoints_topic ON viewpoints(topic);
CREATE INDEX idx_viewpoints_type ON viewpoints(claim_type);

-- 向量索引
CREATE INDEX idx_viewpoints_embedding ON viewpoints 
    USING ivfflat (content_embedding vector_cosine_ops) WITH (lists = 100);
```

#### cases 表（案例表）

```sql
CREATE TABLE cases (
    id VARCHAR(36) PRIMARY KEY,
    kb_id VARCHAR(36) NOT NULL,
    doc_id VARCHAR(36),
    
    -- 案例核心
    case_name VARCHAR(200) NOT NULL,
    industry VARCHAR(100),
    scenario VARCHAR(200),
    
    -- 案例结构
    background TEXT,
    approach TEXT,
    process TEXT,
    result TEXT,
    experience TEXT,                     -- 可复用经验
    lessons TEXT,                        -- 教训
    
    -- 关联
    event_ids JSONB,                     -- 组成案例的事件
    entity_ids JSONB,                    -- 涉及的实体
    
    -- 来源
    source_text TEXT,
    
    -- 向量
    approach_embedding vector(1024),
    experience_embedding vector(1024),
    
    create_time TIMESTAMP,
    is_delete INT DEFAULT 0
);

-- 索引
CREATE INDEX idx_cases_kb ON cases(kb_id);
CREATE INDEX idx_cases_scenario ON cases(scenario);
CREATE INDEX idx_cases_industry ON cases(industry);

-- 向量索引
CREATE INDEX idx_cases_approach_emb ON cases 
    USING ivfflat (approach_embedding vector_cosine_ops) WITH (lists = 100);
CREATE INDEX idx_cases_experience_emb ON cases 
    USING ivfflat (experience_embedding vector_cosine_ops) WITH (lists = 100);
```

### 2.4 数据迁移

现有 `knowledge_chunk` 表中的数据需要重新处理：

```
已入库的 chunk 数据：
├─ 判断 ingestion_mode
│   ├─ deep → 重新提取事件/实体/观点，存入新表 + Neo4j
│   ├─ thinking-model → 已有 thinking_model，不变
│   ├─ standard → 保持 chunk 表，用于描述性知识检索
│   └─ auto → 根据内容重新判断类型
└─ 提供迁移脚本，批量重新入库
```

## 三、Neo4j 图结构

### 3.1 节点类型

```cypher
// Entity 节点（实体）
(:Entity {
    id: 'ent_xxx',
    pg_id: 'PostgreSQL entities 表 ID',
    name: '马斯克',
    type: '人物',
    aliases: ['马斯克', 'Elon Musk'],
    description: '特斯拉CEO',
    facts: [{'fact': '出生于南非'}],
    kb_id: 'kb_xxx'
})

// Event 节点（事件）
(:Event {
    id: 'evt_xxx',
    pg_id: 'PostgreSQL events 表 ID',
    action: '收购',
    result: '成为最大股东',
    event_time: '2022-04-25',
    event_type: '商业行为',
    artifacts: [{'type': '数据', 'name': '440亿美元'}],
    kb_id: 'kb_xxx'
})

// Case 节点（案例）
(:Case {
    id: 'case_xxx',
    pg_id: 'PostgreSQL cases 表 ID',
    name: 'Twitter收购案例',
    industry: '社交媒体',
    kb_id: 'kb_xxx'
})

// Viewpoint 节点（观点）- 可选，也可只在 PostgreSQL
(:Viewpoint {
    id: 'vp_xxx',
    pg_id: 'PostgreSQL viewpoints 表 ID',
    content: '裁员是对的',
    claim_type: 'opinion',
    kb_id: 'kb_xxx'
})
```

### 3.2 关系类型

```cypher
// 实体-事件（带角色）
(Entity)-[:INITIATED {context: '出资440亿', pg_rel_id: 'xxx'}]->(Event)
(Entity)-[:TARGETED]->(Event)
(Entity)-[:PARTICIPATED {role: '财务顾问'}]->(Event)
(Entity)-[:AFFECTED_BY {impact: '失业', degree: 'high'}]->(Event)

// 事件-事件（因果链）
(Event)-[:CAUSED {mechanism: '所有权变更', pg_rel_id: 'xxx'}]->(Event)
(Event)-[:TRIGGERED]->(Event)
(Event)-[:FOLLOWED]->(Event)
(Event)-[:CONCURRENT]->(Event)

// 案例组成
(Event)-[:PART_OF {position: 1}]->(Case)
(Entity)-[:INVOLVED_IN_CASE {role: '主角'}]->(Case)

// 观点关系（可选）
(Viewpoint)-[:EXPRESSED_BY]->(Entity)
(Viewpoint)-[:SUPPORTS]->(Event)
(Viewpoint)-[:OPPOSES]->(Event)

// 实体-实体（社交网络）
(Entity)-[:COLLABORATED_WITH]->(Entity)
(Entity)-[:COMPETES_WITH]->(Entity)
(Entity)-[:ACQUIRED]->(Entity)
```

### 4.2 主入库流程（tpl-main-ingestion）

```
┌─────────────────────────────────────────────────────────────────┐
│                    主入库流程节点设计                            │
├─────────────────────────────────────────────────────────────────┤
│                                                                 │
│  ┌──────────────────────────────────────────────────────────┐  │
│  │  Node 1: DocumentLoader（文档加载器）                     │  │
│  ├──────────────────────────────────────────────────────────┤  │
│  │  类型: ObjectInput                                        │  │
│  │  输入字段:                                                │  │
│  │    ├─ kb_id: 知识库ID                                     │  │
│  │    ├─ doc_id: 文档ID（可选，新建则为空）                   │  │
│  │    ├─ title: 文档标题                                     │  │
│  │    ├─ raw_text: 原始文本                                  │  │
│  │    ├─ source_type: 来源类型（url/file/text/douyin）       │  │
│  │    └─ file_path: 文件路径（可选）                         │  │
│  │  输出: context.variables = {kb_id, doc_id, raw_text, ...} │  │
│  └──────────────────────────────────────────────────────────┘  │
│                         ↓                                      │
│  ┌──────────────────────────────────────────────────────────┐  │
│  │  Node 2: ContentClassifier（内容分类器）                  │  │
│  ├──────────────────────────────────────────────────────────┤  │
│  │  类型: ContentClassifier                                  │  │
│  │  输入: raw_text, model_config_id                          │  │
│  │  输出:                                                    │  │
│  │    ├─ primary_type: 事件密集型/观点密集型/案例型/方法论型 │  │
│  │    ├─ secondary_types: []                                 │  │
│  │    ├─ confidence: 0.85                                    │  │
│  │    └─ classification_result: JSON                         │  │
│  │  节点逻辑: 调用 LLM 判断内容类型                           │  │
│  └──────────────────────────────────────────────────────────┘  │
│                         ↓                                      │
│  ┌──────────────────────────────────────────────────────────┐  │
│  │  Node 3: IngestionRouter（入库路由器）                    │  │
│  ├──────────────────────────────────────────────────────────┤  │
│  │  类型: Router                                             │  │
│  │  输入: classification_result                              │  │
│  │  路由规则:                                                │  │
│  │    ├─ primary_type = '事件密集型' → 调用事件提取子流程    │  │
│  │    ├─ primary_type = '观点密集型' → 调用观点提取子流程    │  │
│  │    ├─ primary_type = '案例型' → 调用案例提取子流程        │  │
│  │    ├─ primary_type = '方法论型' → 调用思维模型入库        │  │
│  │    └─ primary_type = '描述型' → 调用普通RAG入库           │  │
│  │  输出: 子流程执行结果                                      │  │
│  └──────────────────────────────────────────────────────────┘  │
│                         ↓                                      │
│  ┌──────────────────────────────────────────────────────────┐  │
│  │  Node 4: ResultAggregator（结果聚合器）                   │  │
│  ├──────────────────────────────────────────────────────────┤  │
│  │  类型: ResultAggregator                                   │  │
│  │  输入: 子流程执行结果                                      │  │
│  │  输出:                                                    │  │
│  │    ├─ doc_id: 文档ID                                      │  │
│  │    ├─ status: completed/failed                            │  │
│  │    ├─ event_count: 提取的事件数                            │  │
│  │    ├─ entity_count: 提取的实体数                           │  │
│  │    ├─ viewpoint_count: 提取的观点数                        │  │
│  │    └─ case_count: 提取的案例数                             │  │
│  └──────────────────────────────────────────────────────────┘  │
│                                                                 │
└─────────────────────────────────────────────────────────────────┘
```

### 4.3 事件提取子流程（tpl-event-ingestion）

```
┌─────────────────────────────────────────────────────────────────┐
│                    事件提取子流程节点设计                        │
├─────────────────────────────────────────────────────────────────┤
│                                                                 │
│  ┌──────────────────────────────────────────────────────────┐  │
│  │  Node 1: EntityExtractor（实体提取器）                    │  │
│  ├──────────────────────────────────────────────────────────┤  │
│  │  类型: EntityExtractor                                    │  │
│  │  输入: raw_text, kb_id, model_config_id                   │  │
│  │  LLM Prompt: 提取文档中所有实体（人物/组织/产品/地点）    │  │
│  │  输出:                                                    │  │
│  │    ├─ entities: [{name, type, description, aliases}]      │  │
│  │    ├─ entity_count: N                                     │  │
│  │  后续: 存入 entities 表 + 创建 Neo4j Entity 节点          │  │
│  └──────────────────────────────────────────────────────────┘  │
│                         ↓                                      │
│  ┌──────────────────────────────────────────────────────────┐  │
│  │  Node 2: EventExtractor（事件提取器）                     │  │
│  ├──────────────────────────────────────────────────────────┤  │
│  │  类型: EventExtractor                                     │  │
│  │  输入: raw_text, kb_id, model_config_id, entities         │  │
│  │  LLM Prompt: 提取所有事件（含因果关系、附属物）           │  │
│  │  输出:                                                    │  │
│  │    ├─ events: [{subject, action, object, result, time,   │  │
│  │    │            artifacts, cause_event_id}]               │  │
│  │    ├─ event_count: N                                      │  │
│  │  后续: 存入 events 表                                     │  │
│  └──────────────────────────────────────────────────────────┘  │
│                         ↓                                      │
│  ┌──────────────────────────────────────────────────────────┐  │
│  │  Node 3: RelationBuilder（关系构建器）                    │  │
│  ├──────────────────────────────────────────────────────────┤  │
│  │  类型: RelationBuilder                                    │  │
│  │  输入: entities, events                                   │  │
│  │  逻辑:                                                    │  │
│  │    ├─ 构建 Entity-Event 关系（带角色）                    │  │
│  │    ├─ 构建 Event-Event 因果关系                           │  │
│  │    ├─ 存入 entity_event_roles 表                          │  │
│  │    ├─ 存入 event_relations 表                             │  │
│  │  输出:                                                    │  │
│  │    ├─ entity_event_relations: [...]                       │  │
│  │    ├─ event_relations: [...]                              │  │
│  └──────────────────────────────────────────────────────────┘  │
│                         ↓                                      │
│  ┌──────────────────────────────────────────────────────────┐  │
│  │  Node 4: Neo4jWriter（Neo4j 写入器）                      │  │
│  ├──────────────────────────────────────────────────────────┤  │
│  │  类型: Neo4jWriter                                        │  │
│  │  输入: entities, events, relations                        │  │
│  │  逻辑:                                                    │  │
│  │    ├─ CREATE Entity 节点                                  │  │
│  │    ├─ CREATE Event 节点                                   │  │
│  │    ├─ CREATE Entity-[:INITIATED]->Event                   │  │
│  │    ├─ CREATE Event-[:CAUSED]->Event                       │  │
│  │    ├─ 更新 PostgreSQL 表的 neo4j_id 字段                  │  │
│  │  输出:                                                    │  │
│  │    ├─ neo4j_entity_count: N                               │  │
```

### 5.4 Neo4jWriterNodeExecutor

```java
@Component
public class Neo4jWriterNodeExecutor implements FlowNodeExecutor {
    
    @Resource
    private Neo4jClient neo4jClient;
    
    @Resource
    private EntityService entityService;
    
    @Resource
    private EventService eventService;
    
    @Resource
    private EventRelationService eventRelationService;
    
    @Resource
    private EntityEventRoleService entityEventRoleService;
    
    @Override
    public boolean supports(String nodeType) {
        return "Neo4jWriter".equals(nodeType);
    }
    
    @Override
    public FlowNodeExecuteResult execute(FlowNodeDTO node, FlowExecutionContext context) {
        List<String> entityIds = (List<String>) context.getVariable("entity_ids");
        List<String> eventIds = (List<String>) context.getVariable("event_ids");
        String kbId = (String) context.getVariable("kb_id");
        
        // 创建 Entity 节点
        int entityNodeCount = 0;
        for (String entityId : entityIds) {
            Entity entity = entityService.getById(entityId);
            String neo4jId = neo4jClient.createEntityNode(entity);
            entityService.updateNeo4jId(entityId, neo4jId);
            entityNodeCount++;
        }
        
        // 创建 Event 节点
        int eventNodeCount = 0;
        for (String eventId : eventIds) {
            Event event = eventService.getById(eventId);
            String neo4jId = neo4jClient.createEventNode(event);
            eventService.updateNeo4jId(eventId, neo4jId);
            eventNodeCount++;
        }
        
        // 创建 Entity-Event 关系
        List<EntityEventRole> roles = entityEventRoleService.listByEventIds(eventIds);
        int entityEventRelCount = 0;
        for (EntityEventRole role : roles) {
            Entity entity = entityService.getById(role.getEntityId());
            Event event = eventService.getById(role.getEventId());
            String neo4jRelId = neo4jClient.createEntityEventRelation(
                entity.getNeo4jId(), event.getNeo4jId(), role.getRole(), role.getContext()
            );
            entityEventRoleService.updateNeo4jRelId(role.getId(), neo4jRelId);
            entityEventRelCount++;
        }
        
        // 创建 Event-Event 因果关系
        List<EventRelation> relations = eventRelationService.listBySourceIds(eventIds);
        int eventRelCount = 0;
        for (EventRelation rel : relations) {
            Event source = eventService.getById(rel.getSourceEventId());
            Event target = eventService.getById(rel.getTargetEventId());
            String neo4jRelId = neo4jClient.createEventRelation(
                source.getNeo4jId(), target.getNeo4jId(), 
                rel.getRelationType(), rel.getMechanism()
            );
            eventRelationService.updateNeo4jRelId(rel.getId(), neo4jRelId);
            eventRelCount++;
        }
        
        FlowNodeExecuteResult result = FlowNodeExecuteResult.of("success");
        result.getOutput().put("neo4j_entity_count", entityNodeCount);
        result.getOutput().put("neo4j_event_count", eventNodeCount);
        result.getOutput().put("neo4j_entity_event_rel_count", entityEventRelCount);
        result.getOutput().put("neo4j_event_rel_count", eventRelCount);
        return result;
    }
}
```
